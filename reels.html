function handleVideoUpload(event) { const file = event.target.files[0]; if (file) { const videoURL = URL.createObjectURL(file); const player = document.getElementById('reelVideoPlayer'); const placeholder = document.getElementById('reelPlaceholder'); player.src = videoURL; player.style.display = 'block'; // Pastikan video langsung diputar player.play().catch(error => { console.log("Autoplay dicegah browser, klik manual jika perlu:", error); }); placeholder.style.display = 'none'; alert('🎉 Karya Video Berhasil Diunggah!\nSmart Contract aktif dan video Anda resmi terhitung ke dalam jaringan referral.'); } }
